
SELECT customer_id, SUM(amount) total_sales,
       RANK() OVER (ORDER BY SUM(amount) DESC) sales_rank
FROM kifa_transactions
GROUP BY customer_id;


SELECT transaction_date, amount,
       SUM(amount) OVER (ORDER BY transaction_date) running_total
FROM kifa_transactions;


SELECT transaction_date, amount,
       LAG(amount) OVER (ORDER BY transaction_date) previous_amount
FROM kifa_transactions;


SELECT customer_id,
       NTILE(4) OVER (ORDER BY SUM(amount)) customer_quartile
FROM kifa_transactions
GROUP BY customer_id;
